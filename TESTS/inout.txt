begin {
    var n = 1;
    in n;

    array A[n];
    var i = 0;
    while ( i < n) {
        in A[i];
        i = i + 1;
    }

    i = 0;
    j = 1;
    while (i < n) {
        while (j < n) {
            if (A[i] > A[j]) {
                var temp = A[i];
                A[i] = A[j];
                A[j] = temp;
            }
            j = j + 1;
        }
        i = i + 1;
    }

    i = 0;
    while ( i < n) {
        out A[i];
        i = i + 1;
    }

} end
